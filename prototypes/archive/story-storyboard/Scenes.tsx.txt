import Image from "next/image";
import Link from "next/link";
import type { ReactNode } from "react";
import { getPublicSettings } from "@/lib/settings/public";
import { primaryCta } from "@/content/site";
import { services } from "@/content/services";
import { collaboration, processSteps } from "@/content/process";
import { featuredProjects, workGateMet } from "@/content/work";

import { WorkGapNotice } from "@/components/work/WorkGapNotice";

/**
 * The home page's ten-scene story, built directly on the owner's approved storyboard
 * (docs/storyboard-direction.md). Each scene's image is the storyboard panel itself
 * (public/cinema/scene-NN.webp, extracted by scripts/extract_storyboard.py), shown in its
 * original portrait composition. Scene titles and captions are the storyboard's own words.
 *
 * Every scene is readable HTML with its image inline. With motion on, StoryPlayer turns scenes
 * 02–10 into a sticky portrait frame beside the scrolling text, revealing each plate in turn.
 */

type Scene = {
  n: string;
  name: string;
  caption: string;
  /** The storyboard shows an invented brand ("MINERAL"), so the scene is labelled as an illustration. */
  illustration?: boolean;
  /** The plate shows the gold sphere: a soft glint pulses on it (position per scene in app/cinema.css, since the CSP forbids inline styles). */
  glint?: boolean;
};

const SCENES: Scene[] = [
  { n: "01", name: "Origin", caption: "An idea appears — a single point of light in a quiet space.", glint: true },
  { n: "02", name: "Awakening", caption: "The idea creates movement, finding energy in unexpected places.", glint: true },
  { n: "03", name: "Discovery", caption: "New elements enter, revealing different possibilities." },
  { n: "04", name: "Foundation", caption: "The pieces begin to align, establishing a coherent structure.", glint: true },
  { n: "05", name: "Character", caption: "The identity develops its own personality and distinctive character." },
  { n: "06", name: "Identity", caption: "A recognizable signature emerges." },
  { n: "07", name: "Language", caption: "Typography, colors, materials and imagery come together in a unified visual language.", illustration: true },
  { n: "08", name: "Expression", caption: "The identity appears across different formats and experiences.", illustration: true },
  { n: "09", name: "Ecosystem", caption: "Different expressions connect and form a coherent world.", illustration: true },
  { n: "10", name: "Beyond", caption: "A complete identity, always evolving." },
];

const scene = (n: string) => SCENES.find((s) => s.n === n)!;

/**
 * `sizes` per plate placement, from the measured layout (docs/image-spec.md): the hero plate is at
 * most 515 CSS px wide (≥1000 px screens), 420 px on tablets and the column width on phones; the
 * sticky frame is at most 539 px and only shown from 1000 px; inline plates are at most 460 px.
 * Next.js serves AVIF or WebP at the matching width and never above the master's own width.
 */
const SIZES = {
  hero: "(min-width: 1000px) 515px, (min-width: 768px) 420px, calc(100vw - 40px)",
  framed: "539px",
  inline: "(min-width: 768px) 460px, calc(100vw - 40px)",
} as const;

function Plate({ s, priority = false, className = "", place }: { s: Scene; priority?: boolean; className?: string; place: keyof typeof SIZES }) {
  return (
    <figure className={`sb-plate ${className}`} data-plate={s.n}>
      {/* media: scaled by the scroll player · drift: slow ambient camera move (CSS, off for reduced motion) */}
      <div className="sb-plate__media">
        <div className="sb-plate__drift">
          <Image
            src={`/cinema/scene-${s.n}.webp`}
            alt=""
            width={1080}
            height={1805}
            priority={priority}
            quality={85}
            sizes={SIZES[place]}
            className="sb-plate__img"
          />
          {s.glint ? (
            <span className={`sb-plate__glint sb-plate__glint--${s.n}`} aria-hidden="true" />
          ) : null}
        </div>
      </div>
      {s.illustration ? <figcaption className="sb-plate__note">Illustration, not client work</figcaption> : null}
    </figure>
  );
}

function Mark({ s }: { s: Scene }) {
  return (
    <p className="sb-mark">
      <span className="sb-mark__n" aria-hidden="true">
        {s.n}
      </span>
      <span className="sb-mark__name">{s.name}</span>
    </p>
  );
}

/** 01 Origin: the hero. The headline is visible on first paint and never animated in. */
export async function Origin() {
  const { replyTime } = await getPublicSettings();
  const s = scene("01");
  return (
    <section className="sb-hero surface-inverse" aria-labelledby="hero-title" data-scene="01">
      <div className="container-calix sb-hero__grid">
        <div className="sb-hero__copy">
          <Mark s={s} />
          <h1 id="hero-title" className="type-display sb-hero__title">
            Make Your Brand Impossible to Ignore.
          </h1>
          <p className="type-lead measure sb-hero__lead">
            Strategy, design, technology and marketing — connected to build stronger brands and better digital outcomes.
          </p>
          <div className="sb-hero__actions">
            <Link href={primaryCta.href} className="btn btn-primary sb-btn-light" data-cta="hero">
              {primaryCta.label}
            </Link>
            <a href="#work" className="btn btn-secondary">
              Explore Our Work
            </a>
          </div>
          <Fact value={replyTime} render={(v) => <p className="type-small sb-muted">We reply within {v}.</p>} />
          <p className="sb-caption">{s.caption}</p>
        </div>
        <div className="sb-hero__media" data-hero-plate>
          <Plate s={s} priority place="hero" />
        </div>
      </div>
    </section>
  );
}

function Stage({ n, children, id, titleId }: { n: string; children: ReactNode; id?: string; titleId: string }) {
  const s = scene(n);
  return (
    <section className="sb-stage" data-stage={n} id={id} aria-labelledby={titleId}>
      <Plate s={s} className="sb-plate--inline" place="inline" />
      <div className="sb-stage__copy">
        <Mark s={s} />
        <p className="sb-caption">{s.caption}</p>
        {children}
      </div>
    </section>
  );
}

const byKey = (key: string) => services.find((s) => s.key === key)!;

function ServiceBlock({ k, titleId }: { k: string; titleId: string }) {
  const s = byKey(k);
  return (
    <div className="sb-service" id={s.anchor}>
      <p className="sb-kicker">{s.name}</p>
      <h2 id={titleId} className="sb-title">
        {s.heading}
      </h2>
      <p className="measure">{s.body}</p>
      <ul className="sb-list type-small">
        {s.included.map((item) => (
          <li key={item}>{item}</li>
        ))}
      </ul>
      <p className="sb-links">
        <Link href={s.cta.href} className="link font-medium">
          {s.cta.label}
        </Link>
        <Link href={`/services/${s.slug}`} className="link">
          See {s.name}
        </Link>
      </p>
    </div>
  );
}

            Most brands are built in pieces: a logo from one studio, a website from another, campaigns and search from

        <Stage n="03" id="services" titleId="services-title">
          <h2 id="services-title" className="sb-title">
            Five services, planned together.
          </h2>

            {services.map((s, i) => (
