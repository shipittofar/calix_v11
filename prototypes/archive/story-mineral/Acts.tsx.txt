/**
 * The cinematic home page, direction C "Nocturne" (docs/cinematic-direction.md): the seven stages of
 * a brand's evolution spread across the page in dark chapters and ivory pauses.
 *
 * Every scene is complete, readable HTML. Without motion (reduced motion, short screens, no
 * JavaScript, or STORY_PREVIEW=false) each scene is a full-height still in normal flow. With motion,
 * html[data-cinema="on"] and the story marker switch Act I to stacked layers and StoryPlayer drives
 * the transitions with GSAP ScrollTrigger. Images are placeholders (Mineral renders and generated
 * textures in public/cinema) until approved photography replaces them, file for file.
 */

const stage = (id: string) => stages.find((s) => s.id === id)!;

function Mark({ n, label }: { n: string; label: string }) {
  return (
    <p className="cine-mark">
      <span className="cine-mark__n" aria-hidden="true">
        {n}
      </span>
      <span className="cine-mark__label">{label}</span>
    </p>
  );
}

          <h2 id="work-title" className="cine-title" data-lines>

          </h2>

            {differentiation.title} Client projects shown here are published only with the client&apos;s approval, with

          </p>
        </div>
        <div className="col-span-4 mt-12 md:col-span-8 lg:col-span-7 lg:col-start-6 lg:mt-0">
          {workGateMet && featuredProjects.length > 0 ? (
            <>
              <ProjectList projects={featuredProjects} />
              <p className="mt-8">
                <Link href="/work" className="link font-medium">
                  See all work
                </Link>
              </p>
            </>
          ) : (
            <ContentPlaceholder
              title="Featured client projects"
              publicFallback={<WorkGapNotice />}
              needed={[
                "Two to three approved projects (publication level P1 or P2)",
                "Desktop and mobile screenshots, identity files or campaign assets for each",
                "Services delivered, year, and the project objective",
                "Written publication approval reference for each client",
              ]}
            >
              No client work is shown until approved projects are supplied.
            </ContentPlaceholder>
          )}
        </div>
      </div>
    </section>
  );
}
